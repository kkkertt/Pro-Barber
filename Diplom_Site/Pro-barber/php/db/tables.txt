CREATE TABLE bookings (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    phone VARCHAR(20) NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE reviews (
    id SERIAL PRIMARY KEY,
    name VARCHAR(255) NOT NULL,
    text TEXT NOT NULL,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE services (
    id SERIAL PRIMARY KEY, -- Уникальный идентификатор услуги
    name VARCHAR(255) NOT NULL, -- Название услуги (например, "мужская стрижка")
    price INTEGER NOT NULL, -- Цена услуги в рублях (например, 2000)
    description TEXT NOT NULL, -- Описание услуги
    image_url VARCHAR(255), -- Ссылка на изображение услуги (например, "./images/services-1.png")
    duration INTERVAL DEFAULT '00:30:00', -- Продолжительность услуги (по умолчанию 30 минут)
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- Дата создания записи
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP -- Дата последнего обновления записи
);
